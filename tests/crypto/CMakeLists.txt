cmake_minimum_required(VERSION 3.22)
project(dpt_crypto_check LANGUAGES C CXX)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(../../shell/src/main/cpp/external/mbedtls mbedtls)
add_executable(code_item_check code_item_check.cpp ../../shell/src/main/cpp/dpt_crypto.cpp)
target_compile_definitions(code_item_check PRIVATE DPT_HOST_CRYPTO)
target_include_directories(code_item_check PRIVATE ../../shell/src/main/cpp)
target_link_libraries(code_item_check PRIVATE mbedcrypto)
